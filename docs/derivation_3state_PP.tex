% =====================================================================
\section{Plant and Tracking Error}
% =====================================================================

\begin{equation}
  z[k\!+\!1] = z[k]
    + \az[k]\bigl(\fdz[k] + \fTz[k]\bigr)
    + z_D[k]
  \label{eq:plant}
\end{equation}

$\dz[k]\triangleq z_d[k]-z[k]$,\quad
$\dz[k\!+\!1]=\lc\,\dz[k]$

\begin{equation}
  \dz[k\!+\!1]
  = \bigl(z_d[k\!+\!1]\!-\!z_d[k]\bigr)
    + \dz[k] - \az\,\fdz[k] - \az\,\fTz[k] - z_D[k]
  \label{eq:error-dyn}
\end{equation}

% =====================================================================
\section{Control Law}
% =====================================================================

\begin{equation}
  \boxed{\;
    \fdz[k] = \hat{\az}^{-1}[k]\Bigl[
      \bigl(z_d[k\!+\!1]\!-\!z_d[k]\bigr)
      + (1\!-\!\lc)\,\dzhat{3}[k]
      - \hat{z}_D[k]\Bigr]
  \;}
  \label{eq:control}
\end{equation}

Known $\az$, no disturbance:\;
$\fdz[k] = \az^{-1}\bigl[
  (z_d[k\!+\!1]\!-\!z_d[k]) + (1\!-\!\lc)\,\dzhat{3}[k]\bigr]$.

% =====================================================================
\section{True State Dynamics}
% =====================================================================

$\dz_1[k]\!=\!\dz[k\!-\!2]$,\;
$\dz_2[k]\!=\!\dz[k\!-\!1]$,\;
$\dz_3[k]\!=\!\dz[k]$,\;
$\edz[k]\!=\!\dz[k]\!-\!\dzhat{3}[k]$.


\begin{equation}
  \boxed{\;
  \begin{aligned}
    \dz_1[k\!+\!1] &= \dz_2[k] \\
    \dz_2[k\!+\!1] &= \dz_3[k] \\
    \dz_3[k\!+\!1] &= \lc\,\dz_3[k]
      + (1\!-\!\lc)\,\edz[k]
      - \az\,\fTz[k]
  \end{aligned}
  \;}
  \label{eq:true-state}
\end{equation}

% =====================================================================
\section{Estimator}
% =====================================================================

$\dz_m[k]=\dz[k\!-\!2]$,\quad
$e_{z1}[k]=\dz_m[k]-\dzhat{1}[k]$,\quad
$H=\begin{bmatrix}1&0&0\end{bmatrix}$.

\begin{equation}
  \boxed{\;
  \begin{aligned}
    \dzhat{1}[k\!+\!1]
      &= \dzhat{2}[k] + L_1\,e_{z1}[k] \\
    \dzhat{2}[k\!+\!1]
      &= \dzhat{3}[k] + L_2\,e_{z1}[k] \\
    \dzhat{3}[k\!+\!1]
      &= \lc\,\dzhat{3}[k] + L_3\,e_{z1}[k]
  \end{aligned}
  \;}
  \label{eq:estimator}
\end{equation}

% =====================================================================
\section{Error Dynamics}
% =====================================================================

$\bm{e}[k]=[e_{\dz 1},\;e_{\dz 2},\;e_{\dz 3}]^\top$,\quad
$e_{\dz i}[k]=\dz_i[k]-\dzhat{i}[k]$.

\eqref{eq:true-state}$\,-\,$\eqref{eq:estimator}:
\begin{equation}
  \bm{e}[k\!+\!1]
  = (F_e - \bm{L}\,H)\,\bm{e}[k] + \bm{w}[k]
  \label{eq:error-dyn-obs}
\end{equation}

\begin{equation}
  F_e =
  \begin{bmatrix}0&1&0\\0&0&1\\0&0&1\end{bmatrix}
  \label{eq:Fe}
\end{equation}

% =====================================================================
\section{Pole Placement}
% =====================================================================

$F_e - \bm{L}H
=\begin{bmatrix}-L_1&1&0\\-L_2&0&1\\-L_3&0&1\end{bmatrix}$,\quad
all eigenvalues at $\le$:

\begin{align}
  \det(zI\!-\!F_e\!+\!\bm{L}H)
  &= (z\!+\!L_1)\,z\,(z\!-\!1)
     + L_2(z\!-\!1) + L_3 \notag\\
  &= z^3 + (L_1\!-\!1)\,z^2
     + (-L_1\!+\!L_2)\,z
     + (-L_2\!+\!L_3)
  \label{eq:char}
\end{align}

$(z-\le)^3 = z^3 - 3\le\,z^2 + 3\le^2\,z - \le^3$:
\begin{equation}
  \boxed{\;
    L_1 = 1 - 3\le,\quad
    L_2 = 1 - 3\le + 3\le^2,\quad
    L_3 = (1 - \le)^3
  \;}
  \label{eq:gains}
\end{equation}
